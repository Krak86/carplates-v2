// v2: largest car component only, paint cluster chosen from the body band, wrap / bad-mask gates.
// usage: node mask-proto2.mjs <modulesBaseDir> <outDir> <list.json>
import { createRequire } from 'node:module'
import { pathToFileURL } from 'node:url'
import fs from 'node:fs'

const [baseDir, outDir, listFile] = process.argv.slice(2)
const req = createRequire(baseDir + '/package.json')
const sharp = req('sharp')
const { pipeline, RawImage, env } = await import(pathToFileURL(req.resolve('@huggingface/transformers')).href)
env.cacheDir = outDir + '/hf-cache'
const seg = await pipeline('image-segmentation', 'Xenova/segformer-b0-finetuned-ade-512-512', { dtype: 'q8' })

const s2l = c => ((c /= 255) <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4)
const l2s = c => 255 * (c <= 0.0031308 ? c * 12.92 : 1.055 * c ** (1 / 2.4) - 0.055)
const lum = (r, g, b) => 0.2126 * s2l(r) + 0.7152 * s2l(g) + 0.0722 * s2l(b)
function lab(r, g, b) {
  const R = s2l(r), G = s2l(g), B = s2l(b)
  const f = t => (t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116)
  const x = f((0.4124 * R + 0.3576 * G + 0.1805 * B) / 0.95047), y = f(0.2126 * R + 0.7152 * G + 0.0722 * B), z = f((0.0193 * R + 0.1192 * G + 0.9505 * B) / 1.08883)
  return [116 * y - 16, 500 * (x - y), 200 * (y - z)]
}
const hex = h => [1, 3, 5].map(i => parseInt(h.slice(i, i + 2), 16))
const chan0 = (r, n) => { const o = Buffer.alloc(n); for (let i = 0; i < n; i++) o[i] = r.data[i * r.info.channels]; return o }

const items = JSON.parse(fs.readFileSync(listFile, 'utf8'))
const W = 640, K = 5
const results = []
const cells = []

for (const { name, url, target: tgtGiven } of items) {
  try {
    const res = await fetch(url, { headers: { 'User-Agent': 'carsua-proto/0.2 (research)' } })
    if (!res.ok) throw new Error('http ' + res.status)
    const buf = Buffer.from(await res.arrayBuffer())
    const { data, info } = await sharp(buf).resize({ width: W }).removeAlpha().raw().toBuffer({ resolveWithObject: true })
    const { width: w, height: h } = info
    const N = w * h
    const out = await seg(await RawImage.fromBlob(new Blob([await sharp(data, { raw: info }).png().toBuffer()])))
    const car = out.find(o => o.label === 'car')
    let status = 'ok', why = ''
    let wBlur = Buffer.alloc(N), target = '#c4161c', paintLab = [0, 0, 0]
    if (!car) { status = 'skip'; why = 'no car' } else {
      const r = await sharp(Buffer.from(car.mask.data), { raw: { width: car.mask.width, height: car.mask.height, channels: 1 } }).resize(w, h, { kernel: 'cubic' }).raw().toBuffer({ resolveWithObject: true })
      const prob = chan0(r, N)
      // connected components of prob > 127
      const lbl = new Int32Array(N); const comps = []
      const stack = new Int32Array(N)
      for (let s = 0; s < N; s++) {
        if (prob[s] <= 127 || lbl[s]) continue
        const id = comps.length + 1; let sp = 0, area = 0, center = false, x0 = w, x1 = 0, y0 = h, y1 = 0
        stack[sp++] = s; lbl[s] = id
        while (sp) {
          const p = stack[--sp]; area++
          const x = p % w, y = (p / w) | 0
          if (x < x0) x0 = x; if (x > x1) x1 = x; if (y < y0) y0 = y; if (y > y1) y1 = y
          if (x > w * 0.35 && x < w * 0.65 && y > h * 0.35 && y < h * 0.65) center = true
          for (const q of [x > 0 ? p - 1 : -1, x < w - 1 ? p + 1 : -1, y > 0 ? p - w : -1, y < h - 1 ? p + w : -1])
            if (q >= 0 && prob[q] > 127 && !lbl[q]) { lbl[q] = id; stack[sp++] = q }
        }
        comps.push({ id, area, center, x0, x1, y0, y1 })
      }
      comps.sort((a, b) => b.area * (b.center ? 2 : 1) - a.area * (a.center ? 2 : 1))
      const c0 = comps[0]
      if (!c0 || c0.area / N < 0.08) { status = 'skip'; why = 'car too small' } else if (c0.area / N > 0.92) { status = 'skip'; why = 'car too big' } else {
        const bh = c0.y1 - c0.y0 + 1
        const multi = comps.length > 1 && comps[1].area > c0.area * 0.6
        const idx = []
        for (let i = 0; i < N; i++) if (lbl[i] === c0.id) idx.push(i)
        const L = new Float32Array(N * 3)
        for (const i of idx) L.set(lab(data[i * 3], data[i * 3 + 1], data[i * 3 + 2]), i * 3)
        // wrap gate: saturated pixels spread over many hue bins
        const bins = Array(12).fill(0); let sat = 0
        for (const i of idx) { const a = L[i * 3 + 1], b = L[i * 3 + 2]; if (Math.hypot(a, b) > 22) { sat++; bins[Math.floor(((Math.atan2(b, a) + Math.PI) / (2 * Math.PI)) * 12) % 12]++ } }
        const top2 = [...bins].sort((a, b) => b - a).slice(0, 2).reduce((s, v) => s + v, 0)
        const wrap = (sat / idx.length > 0.3 && top2 / Math.max(sat, 1) < 0.6) || (sat / idx.length > 0.15 && top2 / Math.max(sat, 1) < 0.5)
        // k-means on a subsample, init by lightness quantiles
        const sub = idx.filter((_, j) => j % Math.max(1, Math.floor(idx.length / 20000)) === 0)
        const sorted = [...sub].sort((a, b) => L[a * 3] - L[b * 3])
        let cent = Array.from({ length: K }, (_, k) => { const p = sorted[Math.floor(((k + 0.5) / K) * sorted.length)]; return [L[p * 3], L[p * 3 + 1], L[p * 3 + 2]] })
        for (let it = 0; it < 14; it++) {
          const sum = Array.from({ length: K }, () => [0, 0, 0, 0])
          for (const p of sub) {
            let best = 0, bd = 1e9
            for (let k = 0; k < K; k++) { const d = (L[p * 3] - cent[k][0]) ** 2 + (L[p * 3 + 1] - cent[k][1]) ** 2 + (L[p * 3 + 2] - cent[k][2]) ** 2; if (d < bd) { bd = d; best = k } }
            sum[best][0] += L[p * 3]; sum[best][1] += L[p * 3 + 1]; sum[best][2] += L[p * 3 + 2]; sum[best][3]++
          }
          cent = sum.map((s, k) => (s[3] ? [s[0] / s[3], s[1] / s[3], s[2] / s[3]] : cent[k]))
        }
        const assign = p => { let best = 0, bd = 1e9; for (let k = 0; k < K; k++) { const d = (L[p * 3] - cent[k][0]) ** 2 + (L[p * 3 + 1] - cent[k][1]) ** 2 + (L[p * 3 + 2] - cent[k][2]) ** 2; if (d < bd) { bd = d; best = k } } return best }
        // score: area in the body band (10%-62% of bbox height, inner 90% of width), dark clusters penalised (glass, tyres)
        const score = Array(K).fill(0), tot = Array(K).fill(0)
        const bw = c0.x1 - c0.x0 + 1
        const asg = new Uint8Array(N)
        for (const p of idx) {
          const k = assign(p); asg[p] = k; tot[k]++
          const x = p % w, y = (p / w) | 0
          const fy = (y - c0.y0) / bh, fx = (x - c0.x0) / bw
          if (fy > 0.1 && fy < 0.62 && fx > 0.05 && fx < 0.95) score[k]++
        }
        for (let k = 0; k < K; k++) { if (cent[k][0] < 25) score[k] *= 0.5; if (cent[k][0] < 40 && Math.hypot(cent[k][1], cent[k][2]) < 8) score[k] *= 0.7 }
        const pk = score.indexOf(Math.max(...score))
        const c = cent[pk]; paintLab = c
        let ss = 0, sn = 0
        for (const p of idx) if (asg[p] === pk) { ss += (L[p * 3] - c[0]) ** 2 + (L[p * 3 + 1] - c[1]) ** 2 + (L[p * 3 + 2] - c[2]) ** 2; sn++ }
        const sigma = Math.max(7, Math.sqrt(ss / Math.max(sn, 1)) * 1.3)
        const wgt = new Float32Array(N); let ym = 0, yn = 0, strong = 0
        for (const p of idx) {
          const pc = Math.hypot(c[1], c[2]), chroma = Math.hypot(L[p * 3 + 1], L[p * 3 + 2])
          let v
          if (pc > 20) { // coloured paint: match hue + chroma, ignore lightness (shade varies a lot), drop near-black
            const dab = (L[p * 3 + 1] - c[1]) ** 2 + (L[p * 3 + 2] - c[2]) ** 2
            v = Math.exp(-dab / (2 * Math.max(10, sigma * 0.8) ** 2)) * Math.min(1, Math.max(0, (L[p * 3] - 14) / 14))
          } else { // neutral paint: wide lightness band, must stay unsaturated
            const sl = Math.max(16, sigma * 1.5)
            v = Math.exp(-((L[p * 3] - c[0]) ** 2) / (2 * sl * sl)) * Math.exp(-(chroma * chroma) / (2 * 16 * 16))
          }
          wgt[p] = v * Math.min(1, (prob[p] / 255 - 0.5) * 4)
          if (v > 0.6) { strong++; ym += lum(data[p * 3], data[p * 3 + 1], data[p * 3 + 2]); yn++ }
        }
        ym /= Math.max(yn, 1)
        const share = strong / idx.length
        if (wrap) { status = 'skip'; why = 'wrap/multicolour' } else if (Math.hypot(c[1], c[2]) <= 20 && c[0] < 45) { status = 'skip'; why = 'dark neutral paint' } else if (share < 0.12) { status = 'skip'; why = `paint share ${(share * 100).toFixed(0)}%` } else if (multi) { why = 'multi-car' }
        const wb = await sharp(Buffer.from(Uint8Array.from(wgt, v => Math.round(v * 255))), { raw: { width: w, height: h, channels: 1 } }).blur(1.1).raw().toBuffer({ resolveWithObject: true })
        wBlur = chan0(wb, N)
        target = tgtGiven ?? (c[1] > 25 && c[0] < 70 ? '#1f4fb4' : '#c4161c')
        ;(results[results.length] = { name, status, why, share: +share.toFixed(2), paintLab: c.map(v => Math.round(v)), ymean: ym })
        // store for rendering
        results[results.length - 1]._ym = ym
      }
    }
    if (!results.find(r => r.name === name)) results.push({ name, status, why })
    const R = results.find(r => r.name === name)
    // ---- all colours; skip recolour when the photo's own paint is already close ----
    const PAINT = { white: '#f2f2f2', gray: '#7d8085', black: '#17181a', blue: '#1f4fb4', red: '#c4161c', green: '#2e7d32', beige: '#cbb896', brown: '#5b3a24', yellow: '#f2c200', orange: '#f07a10', purple: '#6a2fa0' }
    const dE = (p, q) => Math.hypot((p[0] - q[0]) * 0.6, p[1] - q[1], p[2] - q[2])
    const paintLabs = Object.fromEntries(Object.entries(PAINT).map(([k, v]) => [k, lab(...hex(v))]))
    const med = arr => { arr.sort((x, y) => x - y); return arr[arr.length >> 1] ?? 0 }
    const Ls = [], As = [], Bs = []
    for (let i = 0; i < N; i++) if (wBlur[i] > 140) { const q = lab(data[i * 3], data[i * 3 + 1], data[i * 3 + 2]); Ls.push(q[0]); As.push(q[1]); Bs.push(q[2]) }
    const origLab = [med(Ls), med(As), med(Bs)]
    const origBucket = Object.keys(PAINT).sort((p, q) => dE(origLab, paintLabs[p]) - dE(origLab, paintLabs[q]))[0]
    R.origBucket = origBucket; R.origLab = origLab.map(Math.round)
    const ymean = R._ym ?? 0.5
    const recolour = tgt => {
      const [tr, tg, tb] = hex(tgt).map(s2l)
      const o = Buffer.from(data)
      for (let i = 0; i < N; i++) {
        const al = wBlur[i] / 255
        if (al < 0.01) continue
        const g = lum(data[i * 3], data[i * 3 + 1], data[i * 3 + 2]) / ymean
        const hl = Math.min(1, Math.max(0, (g - 1.6) / 1.4))
        for (let k = 0; k < 3; k++) {
          const t = [tr, tg, tb][k]
          const v = Math.min(1, t * Math.min(g, 3) ** 0.9) * (1 - hl) + hl * Math.min(1, 0.5 + 0.25 * g)
          o[i * 3 + k] = Math.round(data[i * 3 + k] * (1 - al) + l2s(v) * al)
        }
      }
      return o
    }
    fs.mkdirSync(outDir + '/g', { recursive: true })
    const save = (buf, file) => sharp(buf, { raw: info }).resize({ width: 480 }).jpeg({ quality: 82 }).toFile(outDir + '/g/' + file)
    await save(data, name + '__orig.jpg')
    R.tiles = {}
    if (status === 'ok') {
      for (const [k, v] of Object.entries(PAINT)) {
        const d = dE(origLab, paintLabs[k]), same = k === origBucket || d < 13
        R.tiles[k] = { kept: same, d: Math.round(d) }
        await save(same ? data : recolour(v), name + '__' + k + '.jpg')
      }
    }
    console.log(name, status, R.why ?? '', R.share ?? '')
  } catch (e) {
    console.log(name, 'ERROR', e.message)
  }
}
fs.writeFileSync(`${outDir}/v4-results.json`, JSON.stringify(results.map(({ _ym, ...r }) => r), null, 1))
console.log('done', results.filter(r => r.status === 'ok').length, '/', results.length)
